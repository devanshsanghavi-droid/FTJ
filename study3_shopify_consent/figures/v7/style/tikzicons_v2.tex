% tikzicons_v2.tex -- hand-built vector icons for the v2 schematics (loaded by tikzstyle_v2.tex).
% No generative AI anywhere (Elsevier bans AI-generated images in figures): every icon below is a few TikZ paths.
%
% Conventions
%  * Every icon is a pic CENTRED on its position, drawn in mm at a nominal size (given below), stroked 0.6 pt greyD on
%    white (\figLwBox), neutral: icons never carry vendor colour or red. Sandboxes carry the hatch (sandbox only).
%  * Scale with \pic[scale=0.8] ...; line widths do not scale, so strokes stay consistent across icon sizes.
%  * Name a pic to get anchors:  \pic (B) at (20,-10) {ic browser};  then use (B-n), (B-s), (B-e), (B-w), (B-c).
%  * Containers that hold other drawing take a size: {browser={w}{h}}, {sandbox={w}{h}}, {iframe={w}{h}} (mm).
%
% Glyph icons (nominal w x h mm):
%   ic browser 10x8, ic store 10x9, ic cart 10x8, ic banner 12x7 (cookie banner), ic sandbox 8x8, ic worker 8x8
%   (web-worker gear), ic iframe 10x8, ic server 8x9 (ad-vendor server), ic shield 7x8.6, ic lock 7x8.6, ic clock 8x8,
%   ic gate 9.4x5.5 (consent gate: boom barrier), ic plug 9x6 (DevTools probe), ic video 10x7 (video card), ic person 7x8.6 (visitor),
%   ic tag 9x4.6 (test code in the URL)

\tikzset{
  icon/.style={draw=greyD, fill=surface, line width=\figLwBox, line join=round, line cap=round},
  iconline/.style={draw=greyD, line width=\figLwBox, line cap=round, line join=round},
  icondetail/.style={draw=greyM, line width=\figLwBox, line cap=round},
  icondot/.style={fill=greyM, draw=none},
  iconsolid/.style={fill=greyD, draw=none},
}
% \iconanchors{half-w}{half-h}: named anchors (-n -s -e -w -c -ne -nw -se -sw) inside a pic
\newcommand{\iconanchors}[2]{%
  \coordinate (-c) at (0,0);
  \coordinate (-n) at (0,#2); \coordinate (-s) at (0,-#2);
  \coordinate (-e) at (#1,0); \coordinate (-w) at (-#1,0);
  \coordinate (-ne) at (#1,#2); \coordinate (-nw) at (-#1,#2);
  \coordinate (-se) at (#1,-#2); \coordinate (-sw) at (-#1,-#2);}
