% \GAPanel{x}{y}{w}{h}{file}: place a panel PDF at 100 % with its top-left at (x, y) mm and check its size.
% (Like \Panel from tikzstyle_v2.tex, but with no panel letter: a graphical abstract is one picture.)
\newcommand{\GAPanel}[5]{%
  \figCheckInside{GA}{#1}{#2}{#3}{#4}%
  \IfFileExists{#5}{%
    \sbox\figPanelBox{\includegraphics{#5}}%
    \figCheckDim{GA}{width}{\wd\figPanelBox}{#3}%
    \figCheckDim{GA}{height}{\dimexpr\ht\figPanelBox+\dp\figPanelBox\relax}{#4}%
    \node[anchor=north west, inner sep=0pt, outer sep=0pt] at (#1,-#2) {\usebox\figPanelBox};%
  }{%
    \PackageError{ga}{Panel file #5 not found (build the panels first)}{}%
  }}
