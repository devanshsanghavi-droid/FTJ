% Fig 2a (v7.1; binding spec ../../DESIGN_V7.md sections 7 and 8.10, worked example ../../fig1/fig1a_pipeline.tex): the
% sampling flow -- the population, the figure's quiet supporting panel (line art in the sciplot grey chrome; no colour,
% no grounds). A vertical SP_AXIS spine with one box per stage; each box is sized to its text with sciplot's 2.4 x 1.6 mm
% padding and holds the stage name (left) and its count (right) on one row, so the column reads as a ledger; all boxes
% share one width. Every removal between two stages, the first step's included, branches right as a 0.6 pt NEUTRAL line
% with a flat end, its count and reason in 6 pt muted grey, all in one grammar ('197,366 not on Shopify', '80 not
% live', '38 not eligible', ...). FigGenie's nested, labelled regions: three outline-only phase frames named by one
% word each ('Frame', 'Draw', 'Crawl'; 0.6 pt NEUTRAL, names at the top-left in 7 pt muted grey), level 1 only.
% Every stage equals the one above minus its branches (asserted in ../scripts/make_fig2_panels.py). The creator-linked
% stratum is not drawn (outside the Tranco frame; the caption gives it).
% Rows, counts and coordinates: ../build/fig2_layout.tex (\ARows) and ../build/fig2_numbers.tex, both generated by
% ../scripts/make_fig2_panels.py from the registry (no number is typed here). Cell \PanelAW x \FigH mm; the top-left
% 4.5 x 4.5 mm stays empty (panel letter): the frames start 4.9 mm in.
\documentclass[tikz,border=0pt]{standalone}
\def\StyleDir{../../style/}
\input{\StyleDir tikzstyle_v2.tex}

\tikzset{
  every node/.append style={text=SPFG},
  el/.style={draw=SPAXIS, fill=surface, line width=0.6bp, rounded corners=0.5mm},
  region/.style={draw=NEUTRAL, fill=none, line width=0.6bp, rounded corners=0.5mm},
  regionlbl/.style={font=\fsLabel, text=SPMUTED, inner sep=0pt, anchor=base west},
  t/.style={font=\fsLabel, inner sep=0pt},
  sub/.style={font=\fsNote, text=SPMUTED, inner sep=0pt},
  con/.style={-{Stealth[length=3.4bp, width=2.8bp, inset=0.9bp]}, draw=SPAXIS, line width=0.6bp},
  branch/.style={draw=NEUTRAL, line width=0.6bp, line cap=butt},
}
\def\midSeven{0.885}   % half the cap height of 7 pt Arial (mm): centres a 7 pt line in a box
\def\midSix{0.76}      % ... of 6 pt Arial: centres a branch label on its line
% \Region{top}{bottom}{name baseline}{name}: an outline-only phase frame, named at its top-left
\newcommand{\Region}[4]{%
  \draw[region] (\Arx,-#1) rectangle (\ArxR,-#2);
  \node[regionlbl] at (\Arx+2.4,-#3) {#4};}
% \Stage{top}{name}{count}: one stage box (6 mm), name left and count right on one row
\newcommand{\Stage}[3]{%
  \draw[el] (\Abx,-#1) rectangle (\AbxR,-#1-6);
  \node[t, anchor=base west] at (\Abx+2.4,-#1-3-\midSeven) {#2};
  \node[t, anchor=base east] at (\AbxR-2.4,-#1-3-\midSeven) {#3};}
% \Link{bottom of the box above}{top of the box below}: the spine, stopping 0.6 mm short
\newcommand{\Link}[2]{\draw[con] (\Axc,-#1) -- (\Axc,-#2+0.6);}
% \Branch{y}{count and reason}: a removal, flat-ended, its label in 6 pt muted grey
\newcommand{\Branch}[2]{%
  \draw[branch] (\Axc,-#1) -- (\Axlab-1.0,-#1);
  \node[sub, anchor=base west] at (\Axlab,-#1-\midSix) {#2};}
