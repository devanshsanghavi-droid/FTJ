% \PanelBare{x}{y}{w}{h}{file.pdf}: \Panel without the letter (single-panel figures only)
\newcommand{\PanelBare}[5]{%
  \figCheckInside{-}{#1}{#2}{#3}{#4}%
  \sbox\figPanelBox{\includegraphics{#5}}%
  \figCheckDim{-}{width}{\wd\figPanelBox}{#3}%
  \figCheckDim{-}{height}{\dimexpr\ht\figPanelBox+\dp\figPanelBox\relax}{#4}%
  \node[anchor=north west, inner sep=0pt, outer sep=0pt] at (#1,-#2) {\usebox\figPanelBox};}

  \FigureCanvas{35}
  %          x   y    w     h    file
